| Setup for Chipo Django 1 Musicdisk.
|
Set RISCOS$version 3.5
RMEnsure UtilityModule 3.5 Set RISCOS$version 3.1
If "<RISCOS$version>"<>"3.1" Then Error This will only run on Archimedes computers, not RiscPC or later.
|
Set CPU$Type ARM3
RMEnsure Arm3Support 0.0 Set CPU$Type ARM2
|
Set Demo$Dir <Obey$Dir>
|
RMLoad <Demo$Dir>.MemAlloc
FONTSIZE 0K
SPRITESIZE 0K
RMASIZE 0K
SCREENSIZE 40K
RMKILL MemAlloc 
|
If "<CPU$Type>"="ARM2" Then RMLOAD <Demo$Dir>.QTM146d_48 { > null: }
If "<CPU$Type>"<>"ARM2" Then RMLOAD <Demo$Dir>.QTM146d_24 { > null: }
RMLOAD <Demo$Dir>.RasterM28 { > null: }
|
wimpslot -min 586K
<Demo$Dir>.!RunImage
